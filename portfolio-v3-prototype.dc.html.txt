    let sysReduce = false;

    let reduce = false;
    const cl = (v) => Math.max(0, Math.min(1, v));
    const de = document.documentElement;
    this.cur = window.scrollY; this.target = this.cur; this.smooth = false;
    this.onWheel = (e) => {
      if (reduce || e.ctrlKey || de.scrollHeight <= window.innerHeight + 20) return;
      e.preventDefault();
      if (!this.smooth) { this.cur = window.scrollY; this.target = this.cur; }
      const max = de.scrollHeight - window.innerHeight;
      this.target = Math.max(0, Math.min(max, this.target + e.deltaY * (e.deltaMode === 1 ? 40 : 1)));
      this.smooth = true;
    };
    window.addEventListener('wheel', this.onWheel, { passive: false });
    const tick = () => {
      if (this.smooth) {
        this.cur += (this.target - this.cur) * 0.09;
        if (Math.abs(this.target - this.cur) < 0.5) { this.cur = this.target; this.smooth = false; }
        window.scrollTo(0, this.cur);
      }
      const vh = window.innerHeight, vw = window.innerWidth, stick = Math.min(vh, 900);

          const c = el.classList;
          if (c.contains('drift')) return;
          if (c.contains('hz')) {
            const tr = el.querySelector('.track');
            const tx = vw >= 760 && tr ? Math.max(0, tr.scrollWidth - vw) : 0;
            el.style.height = vw >= 760 ? (stick + tx) + 'px' : '';
            el.style.setProperty('--tx', tx + 'px');
          }
          const r = el.getBoundingClientRect();
          let p;
          if (c.contains('pin') || c.contains('hz')) p = cl(-r.top / Math.max(1, r.height - stick));
          else if (c.contains('par')) p = Math.max(-1, Math.min(1, (r.top + r.height / 2 - vh / 2) / vh));
          else if (c.contains('grow')) p = cl((vh - r.top) / (vh * 0.7));
          else if (c.contains('wd')) p = cl((vh * 0.78 - r.top) / (vh * 0.18));
          else if (c.contains('ln')) p = cl((vh * 0.95 - r.top) / (vh * 0.2));
          else if (c.contains('fan-w')) p = cl((vh * 0.6 - r.top) / (vh * 0.45));
          else p = cl((vh - r.top) / (vh * 0.3));
          el.style.setProperty('--p', p.toFixed(3));
        });
      }
      if (root && live) {

        document.querySelectorAll('[data-bg]').forEach((s) => { const r = s.getBoundingClientRect(); if (r.top <= vh * 0.5 && r.bottom > vh * 0.5) pick = s; });
